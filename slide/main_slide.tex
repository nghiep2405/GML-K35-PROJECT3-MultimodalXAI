% Slide 5: Level 1 Methodology
\begin{frame}{Cấp độ 1: Mô hình cơ sở và tăng cường đồ thị GAT}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \textbf{Xây dựng đồ thị tĩnh (Flickr30k Entities):}
      \begin{itemize}
        \item Đỉnh (nodes): Biểu diễn các thực thể và vùng ảnh (bounding boxes).
        \item Cạnh (edges): Quan hệ không gian và đồng xuất hiện ngữ nghĩa giữa các cụm từ.
      \end{itemize}
      \vspace{0.2cm}
      \textbf{Mạng nơ-ron chú ý đồ thị GAT:}
      \begin{itemize}
        \item Cơ chế self-attention học trọng số chú ý $\alpha_{ij}$ phản ánh mức độ quan trọng ngữ nghĩa giữa các đối tượng.
        \item Trích xuất biểu diễn đồ thị $h_{\text{graph}}(v_i)$ qua readout pooling.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Cơ chế kết hợp điểm số}}
        Điểm số cuối cùng kết hợp tuyến tính lồi:
        \begin{equation*}
          S_{\text{final}} = \alpha_{\text{CLIP}} \cdot S_{\text{CLIP}} + (1 - \alpha_{\text{CLIP}}) \cdot S_{\text{graph}}
        \end{equation*}
        \footnotesize
        \textbf{Nguyên tắc nghiêm ngặt:} Trọng số $\alpha_{\text{CLIP}} = 0.6$ được quét và tối ưu trên validation set, đóng băng cố định trước khi suy luận trên test set một lần duy nhất.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 8: Level 2 AdaMF Fusion
\begin{frame}{Cấp độ 2: Dung hợp đặc trưng đa phương thức AdaMF}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \textbf{Cơ chế dung hợp thích nghi AdaMF-based:}
      Thay vì nối ghép thô (concatenation), mô hình tính tương tác động:
      \begin{equation*}
        e_{\text{fused}} = \gamma_T \odot W_T e_{\text{text}} + \gamma_V \odot W_V e_{\text{img}}
      \end{equation*}
      \begin{itemize}
        \item Trọng số $\gamma_T, \gamma_V$ sinh ra từ gating network.
        \item Tự động ưu tiên thông tin thị giác khi ảnh giàu chi tiết và bù đắp bằng ngữ nghĩa văn bản khi ảnh mờ nhạt hoặc trừu tượng.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{\textbf{Xếp hạng lai ghép (Hybrid Reranking)}}
        \begin{equation*}
          S_{\text{hybrid}} = \alpha \cdot S_{\text{CLIP}} + (1 - \alpha) \cdot S_{\text{KGC}}
        \end{equation*}
        \footnotesize
        \textbf{Khóa tham số:} $\alpha_{\text{CLIP}} = 0.2$ trên validation set (ưu tiên 80\% trọng số cho điểm đồ thị tri thức MMKG).
      \end{block}
      \begin{exampleblock}{\textbf{Phạm vi học thuật rõ ràng}}
        \footnotesize
        Triển khai cơ chế dung hợp AdaMF-based phục vụ truy vấn và xếp hạng lại, không tuyên bố tái lập 100\% toàn bộ hệ thống AdaMF-MAT gốc.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cấp độ 2: Phân tích triệt tiêu cấu trúc và hoán đổi cạnh}
  \begin{columns}[c]
    \begin{column}{0.52\textwidth}
      \textbf{Phương pháp luận hoán đổi cạnh (Shuffled Edges):}
      \begin{itemize}
        \item Hoán đổi các đỉnh trong cùng loại quan hệ nhằm bảo toàn degree distribution nhưng phá hủy hoàn toàn ngữ nghĩa bộ ba.
        \item \textbf{Shuffled Edges v1:} Phát hiện có 171 cạnh trùng tập test (tiềm ẩn rò rỉ).
        \item \textbf{Shuffled Edges v2:} Đạt \textbf{overlap 0/0} bằng phương pháp test-aware exclusion $\rightarrow$ Báo cáo hoàn toàn minh bạch.
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{\textbf{Bóc tách giá trị cấu trúc đồ thị}}
        \begin{itemize}
          \item \textbf{Full Graph vs. Shuffled Edges v2:} Chênh lệch $+0.0989$ điểm MRR $\rightarrow$ Khẳng định giá trị thực sự của ngữ nghĩa quan hệ bộ ba.
          \item \textbf{Shuffled Edges v2 vs. No-Edges:} Chênh lệch $+0.0579$ điểm MRR $\rightarrow$ Việc giữ lại degree distribution vẫn mang lại tiên nghiệm cấu trúc nhất định.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 11: Level 3 xAI Framework
\begin{frame}{Cấp độ 3: Khung giải thích xAI dựa trên đồ thị}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \textbf{Nguyên tắc giải thích có tính nhân quả:}
      \begin{itemize}
        \item Giải thích không được là ngụy biện hậu nghiệm (post-hoc rationalization).
        \item Phải phản ánh trực tiếp luồng tính điểm của GAT attention và liên kết bộ ba AdaMF.
      \end{itemize}
      \vspace{0.2cm}
      \textbf{Trích xuất đồ thị con giải thích:}
      \begin{equation*}
        \mathcal{P} = \{ (e_{\text{query}}, r_1, e_1), (e_1, r_2, e_{\text{target}}) \}
      \end{equation*}
      Tối ưu hóa tập cạnh con tối thiểu bảo toàn thứ hạng của thực thể đích.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{\textbf{Bộ tiêu chuẩn định lượng 3 chỉ số}}
        \begin{itemize}
          \item \textbf{Fidelity$^-$ (Tính trung thực qua can thiệp):} Mức sụt giảm điểm số khi can thiệp xóa (mask) các cạnh giải thích.
          \item \textbf{Validity (Tính hợp lệ):} Tỷ lệ đường đi logic hợp lệ, không chứa vòng lặp rỗng.
          \item \textbf{Sparsity (Độ thưa):} Tỷ lệ nén cạnh giải thích so với đồ thị tổng thể ($>85\%$).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% Slide 14: Data Leakage Prevention
\begin{frame}{Phòng chống data leakage và kiểm soát tính tái lập}
  \begin{block}{\textbf{Bốn trụ cột phòng chống data leakage của đồ án}}
    \begin{enumerate}
      \item \textbf{Khóa chữ ký mã băm (Artifact Hash Locking):} Mọi ma trận đặc trưng, tập split và checkpoint được gắn mã SHA-256 bất biến trong \texttt{test\_protocol.json}.
      \item \textbf{Kiểm soát trùng lặp ảnh (Visual Genome Split v2):} Phân cụm pHash ($\le 6$) và SHA-256 đảm bảo các ảnh tương đồng bắt buộc nằm chung một tập train/val/test.
      \item \textbf{Giao thức Frozen Test:} Tập test chỉ được suy luận đúng một lần sau khi siêu tham số $\alpha$ đã được khóa hoàn toàn trên validation set.
      \item \textbf{Kiểm soát chồng lấn cạnh (Edge Overlap Audit):} Đạt \texttt{overlap = 0/0} trên DB15K Shuffled Edges v2; ghi nhận rõ ràng tính chất test-aware exclusion.
    \end{enumerate}
  \end{block}
\end{frame}
