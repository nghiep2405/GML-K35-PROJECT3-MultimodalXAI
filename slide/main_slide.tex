\begin{frame}{CLIP Retrieval Can Struggle with Relational Queries}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{itemize}
        \item CLIP ranks images by \textbf{cosine similarity} of frozen embeddings --- effective, but without explicit graph reasoning.
        \item Images with the same objects but different \textbf{relations} can be difficult to distinguish.
        \item A similarity score alone does not identify the \textbf{evidence} behind a result.
      \end{itemize}
      \vspace{0.15cm}
      \begin{block}{Research question}
        \small Can graph structure (scene graphs, MMKG) improve ranking --- especially for
        \textbf{indirect} queries --- and can the used relations be \textbf{quantified as explanations}?
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \centering
      \resizebox{\linewidth}{!}{%

\begin{frame}{\leveltag{L1Blue}{LEVEL 1} Scene Graph + GAT Rerank CLIP's Top-50}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[
        n/.style={draw=LineGray, fill=SoftBg, rounded corners=2pt, align=center,
                  font=\scriptsize, inner sep=4pt, minimum height=0.8cm, minimum width=2.2cm},
        e/.style={font=\scriptsize\itshape, text=Slate, align=center},
        ar/.style={-{Latex[length=2mm]}, Slate, thick}
      ]
        \node[n, fill=HCMUSBlue, text=white] (img) {image $v_i$\\(CLIP 512-d)};
        \node[n, right=1.3cm of img] (obj) {entity nodes\\(CLIP name 512-d)};
        \node[n, right=1.3cm of obj] (gat) {2-layer GAT\\4 heads $\times$ 32-d};
        \node[n, fill=L1Blue, text=white] at ([xshift=1.3cm]gat.east) (h) {$h_{\rm graph}$\\128-d};
        \draw[ar] (img) -- (obj);
        \draw[ar] (obj) -- (gat);
        \draw[ar] (gat) -- (h);
        \node[e, below=0.3cm of img] {\texttt{contains} edge};
        \node[e, below=0.3cm of obj] {relations + reverse\\edge features 512-d};
        \node[e, below=0.3cm of gat] {query proj.\\512 $\rightarrow$ 128-d};
      \end{tikzpicture}}
      \end{center}
      \vspace{0.15cm}
      \begin{itemize}\footnotesize
        \item Independent graph per image; directed relations + added reverse edges + \texttt{edge\_is\_reverse} flag.
        \item Graph score $S_{\rm graph}=\tilde e_q^{\top} h_{\rm graph}(v_i)$; both scores z-scored per query over the candidate set.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{Score fusion}\footnotesize
        \[
          \scalebox{0.90}{$S_{\rm final}=\alpha_{\rm CLIP}\,z(S_{\rm CLIP})+(1-\alpha_{\rm CLIP})\,z(S_{\rm graph})$}
        \]
        $\alpha_{\rm CLIP}=0.6$ chosen on validation (grid 0--1, step 0.1).
      \end{block}
      \vspace{0.10cm}
      \begin{block}{Training}\footnotesize
        AdamW, lr $10^{-4}$, 100 epochs, batch 8 queries $\times$ (1 positive + 50 hard negatives); 3 seeds; checkpoint = best validation MRR.
      \end{block}
      \vspace{0.10cm}
      {\footnotesize\color{Slate} GAT reranks only CLIP's top-50 --- it cannot rescue a correct image outside that pool.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\leveltag{L1Blue}{LEVEL 1} Most of the Gain Survives Without Real Edges}
  \begin{columns}[t]

      \centering
      \includegraphics[width=0.86\linewidth]{assets/charts/l1_ablation.png}\\[0.02cm]

      \end{itemize}
    \end{column}

\begin{frame}{\leveltag{L2Orange}{LEVEL 2} MMKG Adds Relations Beyond the Image}
  \begin{columns}[c]
    \begin{column}{0.50\textwidth}
      \begin{itemize}\footnotesize
        \item Level-1 graphs are \textbf{per-image} --- they cannot bridge entities across knowledge.
        \item \textbf{Indirect query}: anchor $e$ and relation $r$ given; rank images of the entity filling $(e, r, ?)$ or $(?, r, t)$.
        \item Entity representation $=$ fused \textbf{structural + text + visual} features (AdaMF attention over modalities; missing modality masked before softmax).
        \item Scoring: \textbf{RotatE} triple score; MAT adversarial term disabled ($\mu=0$).
      \end{itemize}
      \vspace{0.1cm}
      \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[
        m/.style={draw=LineGray, fill=SoftBg, rounded corners=2pt, font=\scriptsize,
                  align=center, inner sep=3pt, minimum height=0.62cm},
        ar/.style={-{Latex[length=2mm]}, Slate}
      ]
        \node[m] (t) {text feat.};
        \node[m, right=0.25cm of t] (v) {visual feat.};
        \node[m, right=0.25cm of v] (s) {structural\\embedding};
        \node[m, fill=L2Orange, text=white, right=0.55cm of s] (f) {AdaMF\\fusion};
        \node[m, right=0.25cm of f] (r) {RotatE\\score};
        \draw[ar] (t) -- (f); \draw[ar] (v) -- (f); \draw[ar] (s) -- (f);
        \draw[ar] (f) -- (r);
      \end{tikzpicture}}
      \end{center}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Datasets (shared retrieval framework)}\footnotesize
        \textbf{DB15K}: 12{,}842 entities, 279 relations, 25{,}925 images; clean split 71{,}300 / 7{,}922 / official test.\\[0.12cm]
        \textbf{FB15K-237-IMG}: 14{,}541 entities, 237 relations, 139{,}814 images; official KG split + image split (seed 2026).
      \end{block}
      \vspace{0.15cm}
      \begin{block}{Query groups}\footnotesize
        \textbf{Direct} = entity name given (CLIP already strong).\\
        \textbf{Indirect} = relation-mediated; 2{,}282/3{,}195 (DB15K) and 8{,}592/24{,}564 (FB15K) direct/indirect test queries.
      \end{block}
      \vspace{0.15cm}
      {\footnotesize\color{Slate} Method implemented from AdaMF-MAT (LREC-COLING 2024), adapted to image retrieval.}
    \end{column}
  \end{columns}
\end{frame}

  \end{center}
  \vspace{0.3cm}
  \begin{columns}[t]
    \begin{column}{0.32\textwidth}\footnotesize
      \textbf{Intervention-based attribution}: evidence is selected by changes in the ranking score, rather than path plausibility alone. GAT uses direct masking; AdaMF uses a local relearning approximation.
    \end{column}
    \begin{column}{0.32\textwidth}\footnotesize
      \textbf{Metrics}\\
      \textbf{Fid+}: relative softmax-probability drop when $S$ is removed (vs.\ random).\\
      \textbf{Fid$-$}: relative probability drop when only $S$ is kept.\\
      \textbf{PN/PS}: top-1 lost / kept.\\
      \textbf{Sparsity}: fraction of units discarded.\\
      \textbf{Structural validity}: elements legal in the graph.
    \end{column}
    \begin{column}{0.32\textwidth}\footnotesize
      \textbf{$K^\star$ selection}\\Chosen on validation: smallest $K$ whose median completeness $\ge 0.8$.\\[0.15cm]

\begin{frame}[noframenumbering]{Backup: Metrics and Evaluation Protocol}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Retrieval metrics}\footnotesize
        \textbf{Recall@K}: fraction of queries with $\ge 1$ relevant image in top-$K$.\\[0.1cm]
        \textbf{MRR}: mean reciprocal rank of the best relevant image; optimistic ranking (count strictly greater scores).\\[0.1cm]
        \textbf{Direct / Indirect MRR}: computed separately on each query group (assignment \S 7).
      \end{block}
      \vspace{0.15cm}
      \begin{block}{Seeds \& selection}\footnotesize
        3 seeds for full models (mean $\pm$ sample SD, $n{-}1$); ablations 1 run (flagged). $\alpha$, checkpoint, $K^\star$ --- validation only; test evaluated once.
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Leakage controls}\footnotesize
        \begin{itemize}
          \item VG: SHA-256 + pHash clustering \emph{before} split --- near-duplicates stay in one split.
          \item DB15K: official validation discarded (2{,}423 duplicate rows); clean 90/10 re-split (seed 2027); official test byte-identical.
          \item FB15K: image clusters hashed before split (seed 2026); cross-entity duplicates dropped.
          \item Entity visual features averaged over \emph{train} images only.
          \item Shuffled-edge controls exclude val/test collisions (disclosed, not fully blind).
          \item ViHeritage: text-masking re-run for name leakage (38.9\% have the anchor name in target-entity text).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
